    UI Architecture Refactor — Implementation Plan#

    For Claude: REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

    Goal: Restructure the Kariana Assistant UI code so that any AI session can quickly find "where do I change X?" — without visual/UX changes.

    Architecture: Central ThemeManager singleton owns all 72+ colors in a TMap, loaded from JSON with C++ fallback defaults. Semantic tokens map role names to raw colors. Three focused StyleSets replace one monolithic set. Layout constants replace ~195 magic numbers. Sub-controllers decompose the panel hub.

    Tech Stack: UE5.7 Slate, C++, JSON (themes.json), FSlateStyleSet, FSlateStyleRegistry

    Reference docs: docs/internal/kariana-ui-architecture/ (14 files), docs/plans/ui-refactor/ (6 phase files)

    Module: UnrealAssistantIntegration (Source/UnrealAssistantIntegration/)

    Build/test: User builds in Rider, reloads UE. No automated unit tests for Slate widgets — verification is compile + visual inspection in the running editor.

void FKarianaThemeManager::LoadTheme(const FString& ThemeName)
{
    CurrentThemeName = ThemeName;

    // Resolve plugin root and find themes.json
    FString PluginRoot;
    TSharedPtr<IPlugin> Plugin = IPluginManager::Get().FindPlugin(TEXT("KARIANA"));
    if (Plugin.IsValid())
    {
        PluginRoot = Plugin->GetBaseDir();
    }

    FString JsonPath = FPaths::Combine(PluginRoot, TEXT("Content/Config/themes.json"));

    FString JsonString;
    if (!FFileHelper::LoadFileToString(JsonString, *JsonPath))
    {
        UE_LOG(LogTemp, Warning, TEXT("KarianaThemeManager: themes.json not found at %s — using defaults"), *JsonPath);
        return;
    }

    TSharedPtr<FJsonObject> RootObject;
    TSharedRef<TJsonReader<>> Reader = TJsonReaderFactory<>::Create(JsonString);
    if (!FJsonSerializer::Deserialize(Reader, RootObject) || !RootObject.IsValid())
    {
        UE_LOG(LogTemp, Warning, TEXT("KarianaThemeManager: Failed to parse themes.json — using defaults"));
        return;
    }

    const TSharedPtr<FJsonObject>* ThemeObject;
    if (!RootObject->TryGetObjectField(ThemeName, ThemeObject))
    {
        UE_LOG(LogTemp, Warning, TEXT("KarianaThemeManager: Theme '%s' not found — using defaults"), *ThemeName);
        return;
    }

    for (const auto& Pair : (*ThemeObject)->Values)
    {
        const TArray<TSharedPtr<FJsonValue>>* ArrayValue;
        if (Pair.Value->TryGetArray(ArrayValue) && ArrayValue->Num() >= 3)
        {
            float R = (*ArrayValue)[0]->AsNumber();
            float G = (*ArrayValue)[1]->AsNumber();
            float B = (*ArrayValue)[2]->AsNumber();
            float A = ArrayValue->Num() >= 4 ? (*ArrayValue)[3]->AsNumber() : 1.0f;
            Colors.Add(FName(*Pair.Key), FLinearColor(R, G, B, A));
        }
    }

    UE_LOG(LogTemp, Log, TEXT("KarianaThemeManager: Loaded theme '%s' (%d colors)"), *ThemeName, Colors.Num());
}

FLinearColor FKarianaThemeManager::GetColor(FName Key) const
{
    const FLinearColor* Found = Colors.Find(Key);
    if (Found)
    {
        return *Found;
    }
    UE_LOG(LogTemp, Warning, TEXT("KarianaThemeManager: Color '%s' not found"), *Key.ToString());
    return FLinearColor(1.0f, 0.0f, 1.0f); // Magenta = missing color
}

FLinearColor FKarianaThemeManager::GetToken(FName TokenName) const
{
    const FName* ColorKey = TokenAliases.Find(TokenName);
    if (ColorKey)
    {
        return GetColor(*ColorKey);
    }
    // Fall back to direct color lookup
    return GetColor(TokenName);
}

    Files: - Create: Source/UnrealAssistantIntegration/Public/Theme/KarianaUIState.h - Modify: Source/UnrealAssistantIntegration/Private/Widgets/SKarianaChatPanel.h - Modify: Source/UnrealAssistantIntegration/Private/Widgets/SKarianaChatPanel.cpp

    Problem: bIsProcessing is duplicated in SKarianaChatInput, SKarianaContextBar, and SKarianaStatusOverlay (orphaned). The panel pushes SetProcessing() to each child manually from OnStreamingStateChanged. Easy to miss a widget, state can drift.

    Step 1: Create the shared state struct

    // KarianaUIState.h
#pragma once
#include "CoreMinimal.h"

DECLARE_MULTICAST_DELEGATE(FOnKarianaUIStateChanged);

/**
 * Shared observable UI state. Widgets subscribe to OnChanged
 * instead of being push-updated by the panel.
 */
struct UNREALASSISTANTINTEGRATION_API FKarianaUIState
{
    // Processing
    bool bIsProcessing = false;
    bool bIsStreaming = false;

    // Connection
    bool bIsConnected = false;
    bool bIsConnecting = false;

    // Mode
    EKarianaAssistantMode CurrentMode = EKarianaAssistantMode::Validate;

    // Notification delegate
    FOnKarianaUIStateChanged OnChanged;

    void SetProcessing(bool bNewProcessing)
    {
        if (bIsProcessing != bNewProcessing)
        {
            bIsProcessing = bNewProcessing;
            OnChanged.Broadcast();
        }
    }

    void SetStreaming(bool bNewStreaming)
    {
        if (bIsStreaming != bNewStreaming)
        {
            bIsStreaming = bNewStreaming;
            OnChanged.Broadcast();
        }
    }

    void SetConnected(bool bConnected, bool bConnecting_)
    {
        if (bIsConnected != bConnected || bIsConnecting != bConnecting_)
        {
            bIsConnected = bConnected;
            bIsConnecting = bConnecting_;
            OnChanged.Broadcast();
        }
    }
};
