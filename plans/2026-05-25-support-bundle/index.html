    Support Bundle Implementation Plan#

    For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

    Goal: Give a Kariana client a one-action, auto-redacted .zip they can email to hello@kariana.ai (or drop on a USB) so we can troubleshoot — produced by a standalone script that works even when the editor/daemon is broken.

    Architecture: A pure-Python, stdlib-only support package under Content/Python/support/ collects versions, system info, redacted log tails, and safe configs, then writes summary.txt + manifest.json into a timestamped zip. Both a standalone CLI (kariana-support.py) and an in-editor MCP tool (handlers/support_commands.py) call the same support.bundle.create_bundle(). Secret-scrubbing patterns live in support/redact.py; memory/privacy.py is refactored to import them (single source of truth, no duplicated regex).

    Tech Stack: Python 3 (stdlib only — re, json, zipfile, pathlib, subprocess, socket, ctypes, platform, hashlib, argparse), pytest.

    Scope note: This plan covers subsystem A — the Python bundle library, standalone CLI, and MCP tool. The in-editor C++ Slate "Report a problem" button (subsystem B) is deferred to a follow-up plan that requires exploring SKarianaChatPanel. The standalone CLI + MCP tool deliver a working, shippable feature on their own.

    Spec: docs/plans/2026-05-25-support-bundle-design.md

"""Canonical secret-scrubbing patterns for KARIANA.

Single source of truth: memory/privacy.py imports SECRET_PATTERNS from here so
the regex set is defined exactly once.
"""

import re
from typing import List, Pattern

REDACTION_PLACEHOLDER = "[REDACTED]"

SECRET_PATTERNS: List[Pattern] = [
    re.compile(r'(?i)(api[_-]?key|apikey)\s*[=:]\s*["\']?[\w\-]+'),
    re.compile(r'(?i)(secret|password|passwd|pwd)\s*[=:]\s*["\']?[\w\-]+'),
    re.compile(r'(?i)(token|auth|bearer)\s*[=:]\s*["\']?[\w\-]+'),
    re.compile(r'(?i)(aws|gcp|azure)[_-]?(access|secret)?[_-]?key\s*[=:]\s*["\']?[\w\-]+'),
    re.compile(r'sk-[a-zA-Z0-9]{20,}'),            # OpenAI API keys
    re.compile(r'ghp_[a-zA-Z0-9]{36}'),            # GitHub tokens
    re.compile(r'xoxb-[a-zA-Z0-9\-]+'),            # Slack tokens
    # AWS API Gateway / Bedrock endpoints
    re.compile(r'(?i)https?://[a-z0-9\-]+\.execute-api\.[a-z0-9\-]+\.amazonaws\.com\S*'),
    re.compile(r'(?i)(kariana_bedrock_\w+|bedrock[_-]?url)\s*[=:]\s*\S+'),
]


def redact_text(text: str) -> str:
    """Replace every secret-pattern match with REDACTION_PLACEHOLDER.

    Fail-safe by construction: anything matching a pattern is removed. Returns
    the input unchanged when it is empty or None.
    """
    if not text:
        return text
    result = text
    for pattern in SECRET_PATTERNS:
        result = pattern.sub(REDACTION_PLACEHOLDER, result)
    return result

"""Resolve on-disk locations for the support bundle.

Mirrors the paths the daemon uses so the standalone collector finds the same
files. Hard-excluded secret files are listed here, never read by the collector.
"""

import os
from pathlib import Path
from typing import List, Optional

# Files that must NEVER enter a bundle (license token + Bedrock creds).
EXCLUDED_FILENAMES = frozenset({".kariana_token", ".kariana_config"})


def user_data_dir() -> Path:
    """Per-user KARIANA data dir, matching the daemon's resolution order."""
    if os.name == "nt":
        local = os.environ.get("LOCALAPPDATA")
        if local:
            return Path(local) / "KARiANA"
        return Path.home() / "AppData" / "Local" / "KARiANA"
    return Path.home() / ".config" / "KARiANA"


def models_json_path() -> Path:
    """User model-catalog override (safe to share)."""
    return user_data_dir() / "models.json"


def admin_config_path() -> Path:
    """Gradio admin dashboard config (tool/model config, no secrets)."""
    return Path(__file__).resolve().parent.parent / "admin_dashboard" / "config.json"


def find_project_dir(start: Optional[Path] = None) -> Optional[Path]:
    """Walk up from `start` (or this file's location) to the dir holding a
    *.uproject file. Returns None if none found."""
    current = (start or Path(__file__).resolve()).resolve()
    if current.is_file():
        current = current.parent
    for candidate in [current, *current.parents]:
        try:
            if any(candidate.glob("*.uproject")):
                return candidate
        except OSError:
            continue
    return None


def daemon_log_path(project_dir: Path) -> Path:
    """KarianaDaemon.log under the project's Saved/Logs."""
    return project_dir / "Saved" / "Logs" / "KarianaDaemon.log"


def unreal_log_paths(project_dir: Path) -> List[Path]:
    """All *.log files under the project's Saved/Logs (UE engine logs)."""
    logs_dir = project_dir / "Saved" / "Logs"
    if not logs_dir.is_dir():
        return []
    return sorted(logs_dir.glob("*.log"))

"""Collect diagnostic data for a support bundle.

Pure stdlib. Every function is safe to call when the editor/daemon is down.
Top-level failures are surfaced to the caller (bundle.py) which aggregates them
into manifest.collection_errors; sub-probes (GPU/RAM) degrade to 'unknown'.
"""

import json
import os
from pathlib import Path
from typing import Dict, Optional

from support import paths
from support.redact import redact_text


def collect_versions() -> dict:
    """Plugin / build / engine version from KARIANA.uplugin (best effort)."""
    try:
        from utils.plugin_version import get_plugin_version
        return dict(get_plugin_version())
    except Exception as exc:  # pragma: no cover - exercised via bundle error path
        return {"plugin_version": "unknown", "build_number": 0,
                "engine_version": "unknown", "error": str(exc)}


def read_log_tail(path: Path, max_bytes: int = 5 * 1024 * 1024) -> str:
    """Return the last `max_bytes` of a log file, decoding leniently.
    Returns '' if the file is missing or unreadable."""
    try:
        if not path.is_file():
            return ""
        size = path.stat().st_size
        with open(path, "rb") as fh:
            if size > max_bytes:
                fh.seek(size - max_bytes)
            data = fh.read()
        return data.decode("utf-8", errors="replace")
    except OSError:
        return ""


def collect_logs(project_dir: Optional[Path], max_bytes: int = 5 * 1024 * 1024) -> Dict[str, str]:
    """Redacted tails of the daemon log + UE project logs, keyed by filename."""
    if project_dir is None:
        return {}
    out: Dict[str, str] = {}
    for log_path in paths.unreal_log_paths(project_dir):
        tail = read_log_tail(log_path, max_bytes)
        if tail:
            out[log_path.name] = redact_text(tail)
    return out


def collect_safe_configs() -> Dict[str, str]:
    """Read known-safe config files (no secrets). Skips missing files."""
    out: Dict[str, str] = {}
    for name, path in (("models.json", paths.models_json_path()),
                       ("admin_config.json", paths.admin_config_path())):
        try:
            if path.is_file() and path.name not in paths.EXCLUDED_FILENAMES:
                out[name] = redact_text(path.read_text(encoding="utf-8", errors="replace"))
        except OSError:
            continue
    return out

def parse_nvidia_smi(output: str) -> list:
    """Parse `nvidia-smi --query-gpu=name,memory.total,driver_version
    --format=csv,noheader` output into a list of dicts."""
    gpus = []
    for line in output.strip().splitlines():
        parts = [p.strip() for p in line.split(",")]
        if len(parts) == 3 and parts[0]:
            gpus.append({"name": parts[0], "memory_total": parts[1],
                         "driver_version": parts[2]})
    return gpus


def _gpu_info() -> list:
    """GPU details via nvidia-smi, falling back to Windows CIM. '[]' on failure."""
    try:
        res = subprocess.run(
            ["nvidia-smi", "--query-gpu=name,memory.total,driver_version",
             "--format=csv,noheader"],
            capture_output=True, text=True, timeout=10,
        )
        if res.returncode == 0:
            parsed = parse_nvidia_smi(res.stdout)
            if parsed:
                return parsed
    except (OSError, subprocess.SubprocessError):
        pass
    if os.name == "nt":
        try:
            res = subprocess.run(
                ["powershell", "-NoProfile", "-Command",
                 "Get-CimInstance Win32_VideoController | "
                 "Select-Object Name,AdapterRAM,DriverVersion | ConvertTo-Json"],
                capture_output=True, text=True, timeout=15,
            )
            if res.returncode == 0 and res.stdout.strip():
                data = json.loads(res.stdout)
                if isinstance(data, dict):
                    data = [data]
                return [{"name": d.get("Name"),
                         "memory_total": d.get("AdapterRAM"),
                         "driver_version": d.get("DriverVersion")} for d in data]
        except (OSError, subprocess.SubprocessError, json.JSONDecodeError):
            pass
    return []


def _ram_gb() -> float:
    """Total physical RAM in GB, or 0.0 if undeterminable."""
    try:
        if os.name == "nt":
            class _MemStatus(ctypes.Structure):
                _fields_ = [("dwLength", ctypes.c_ulong),
                            ("dwMemoryLoad", ctypes.c_ulong),
                            ("ullTotalPhys", ctypes.c_ulonglong),
                            ("ullAvailPhys", ctypes.c_ulonglong),
                            ("ullTotalPageFile", ctypes.c_ulonglong),
                            ("ullAvailPageFile", ctypes.c_ulonglong),
                            ("ullTotalVirtual", ctypes.c_ulonglong),
                            ("ullAvailVirtual", ctypes.c_ulonglong),
                            ("sullAvailExtendedVirtual", ctypes.c_ulonglong)]
            stat = _MemStatus()
            stat.dwLength = ctypes.sizeof(_MemStatus)
            ctypes.windll.kernel32.GlobalMemoryStatusEx(ctypes.byref(stat))
            return round(stat.ullTotalPhys / (1024 ** 3), 1)
        pages = os.sysconf("SC_PHYS_PAGES")
        page_size = os.sysconf("SC_PAGE_SIZE")
        return round(pages * page_size / (1024 ** 3), 1)
    except (OSError, ValueError, AttributeError):
        return 0.0


def _daemon_running(port: int = 9877) -> bool:
    try:
        with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as sock:
            sock.settimeout(0.5)
            return sock.connect_ex(("127.0.0.1", port)) == 0
    except OSError:
        return False


def _active_backend() -> str:
    """Best-effort active backend from admin config; 'unknown' otherwise."""
    try:
        cfg = paths.admin_config_path()
        if cfg.is_file():
            data = json.loads(cfg.read_text(encoding="utf-8", errors="replace"))
            return str(data.get("active_profile") or data.get("backend") or "unknown")
    except (OSError, json.JSONDecodeError):
        pass
    return "unknown"


def _free_disk_gb(project_dir: Optional[Path]) -> float:
    try:
        target = project_dir if project_dir and project_dir.exists() else Path.home()
        return round(shutil.disk_usage(str(target)).free / (1024 ** 3), 1)
    except OSError:
        return 0.0

"""Assemble a redacted support bundle zip.

Orchestrates the collector, builds a safe-by-construction summary + manifest,
and writes a single timestamped zip. Never aborts: each collection step is
wrapped and failures are recorded in manifest.collection_errors.
"""

import hashlib
import json
import uuid
import zipfile
from datetime import datetime
from pathlib import Path

from support import collector, paths
from support.redact import redact_text

SUPPORT_EMAIL = "hello@kariana.ai"


def make_support_id(timestamp: str, machine_id: str) -> str:
    """Stable 8-hex-char id from timestamp + machine id (for correlation)."""
    digest = hashlib.sha256(f"{timestamp}{machine_id}".encode("utf-8")).hexdigest()
    return digest[:8]


def extract_top_errors(logs: dict, limit: int = 20) -> List[str]:
    """Pull error/exception lines from redacted logs, re-redact, cap count/length."""
    keywords = ("error", "exception", "fatal", "traceback")
    found: List[str] = []
    for content in logs.values():
        for line in content.splitlines():
            low = line.lower()
            if any(k in low for k in keywords):
                found.append(redact_text(line.strip())[:300])
                if len(found) >= limit:
                    return found
    return found


def build_summary(data: dict) -> str:
    """Human-readable, ALLOWLIST-ONLY summary. Reads only curated fields — never
    raw log bodies — so it is clean even if the log scrubber misses something."""
    v = data.get("versions", {})
    s = data.get("system_info", {})
    gpu_names = ", ".join(g.get("name", "?") for g in s.get("gpus", [])) or "unknown"
    lines = [
        "KARIANA Support Bundle",
        "=" * 40,
        f"Email this entire .zip to {SUPPORT_EMAIL}",
        "(or hand it to your Kariana contact).",
        f"Support ID: {data.get('support_id', 'unknown')}",
        f"Generated:  {data.get('timestamp', 'unknown')}",
        "",
        "Versions",
        "--------",
        f"  Plugin:  {v.get('plugin_version', 'unknown')} (build {v.get('build_number', '?')})",
        f"  Engine:  {v.get('engine_version', 'unknown')}",
        "",
        "System",
        "------",
        f"  OS:      {s.get('os', 'unknown')}",
        f"  CPU:     {s.get('cpu', 'unknown')} ({s.get('cpu_count', '?')} cores)",
        f"  RAM:     {s.get('ram_gb', '?')} GB",
        f"  GPU:     {gpu_names}",
        f"  Disk:    {s.get('free_disk_gb', '?')} GB free",
        f"  Python:  {s.get('python_version', 'unknown')}",
        f"  Backend: {s.get('active_backend', 'unknown')}",
        f"  Daemon:  {'running' if s.get('daemon_running') else 'not running'}",
        "",
        "Recent errors (redacted)",
        "------------------------",
    ]
    top = data.get("top_errors", [])
    lines.extend(f"  {e}" for e in top) if top else lines.append("  (none captured)")
    errs = data.get("collection_errors", [])
    if errs:
        lines += ["", "Collection warnings", "-------------------"]

    def _safe(step_name: str, fn, errors: List[str], default):
    """Run a collector step; on failure record it and return `default`."""
    try:
        return fn()
    except Exception as exc:  # noqa: BLE001 - bundle must never abort
        errors.append(f"{step_name}: {exc}")
        return default


def create_bundle(output_dir: Optional[Path] = None,
                  project_dir: Optional[Path] = None,
                  audit_log_json: Optional[str] = None,
                  now: Optional[datetime] = None) -> Path:
    """Collect → redact → write summary + manifest → zip. Returns the zip path.

    Always produces a (possibly partial) zip. `output_dir` defaults to the
    Desktop. `project_dir` is auto-detected when None.
    """
    now = now or datetime.now()
    timestamp = now.strftime("%Y%m%d-%H%M%S")
    machine_id = f"{uuid.getnode():x}"
    support_id = make_support_id(timestamp, machine_id)

    if project_dir is None:
        project_dir = paths.find_project_dir()

    errors: List[str] = []
    versions = _safe("versions", collector.collect_versions, errors, {})
    system_info = _safe("system_info",
                        lambda: collector.collect_system_info(project_dir), errors, {})
    logs = _safe("logs", lambda: collector.collect_logs(project_dir), errors, {})
    configs = _safe("configs", collector.collect_safe_configs, errors, {})

    data = {
        "support_id": support_id,
        "timestamp": timestamp,
        "versions": versions,
        "system_info": system_info,
        "logs": logs,
        "top_errors": extract_top_errors(logs),
        "collection_errors": errors,
    }

    included: List[str] = ["summary.txt", "manifest.json"]
    included += [f"logs/{name}" for name in logs]
    included += [f"config/{name}" for name in configs]
    if audit_log_json:
        included.append("telemetry.json")
    data["included_files"] = included

    if output_dir is None:
        desktop = Path.home() / "Desktop"
        output_dir = desktop if desktop.is_dir() else Path.home()
    output_dir.mkdir(parents=True, exist_ok=True)
    zip_path = output_dir / f"kariana-support-{timestamp}.zip"

    with zipfile.ZipFile(zip_path, "w", zipfile.ZIP_DEFLATED) as zf:
        zf.writestr("summary.txt", build_summary(data))
        zf.writestr("manifest.json", json.dumps(build_manifest(data), indent=2))
        for name, content in logs.items():
            if name not in paths.EXCLUDED_FILENAMES:
                zf.writestr(f"logs/{name}", content)
        for name, content in configs.items():
            if name not in paths.EXCLUDED_FILENAMES:
                zf.writestr(f"config/{name}", content)
        if audit_log_json:
            zf.writestr("telemetry.json", redact_text(audit_log_json))

    return zip_path
